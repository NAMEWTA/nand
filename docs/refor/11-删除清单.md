```bash

  --exclude-dir=tmp --exclude-dir=refor \
  | grep -v -e "src/modules/icons/core/res/" -e "scripts/bundle-budget.json" -e "test/release/documentation-policy.test.ts"

# 浏览器扩展

# 旧界面（阶段 9 之后）
```
